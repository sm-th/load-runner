load-runner.toml
[bus]
kind = "sqlite"

[runner]
name = "train"
command = "python train.py --lr {lr}"
wait_for_reply = true

[params]
lr = 0.3
epochs = 3

[agent]
command = "claude -p"

load-runner run # mechanics
load-runner agent # intelligence
load-runner watch # people
